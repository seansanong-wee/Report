        // Tab switching logic
        function switchTab(tabName) {
            const submitTab = document.getElementById('tab-submit');
            const dashboardTab = document.getElementById('tab-dashboard');
            const navSubmit = document.getElementById('nav-submit');
            const navDashboard = document.getElementById('nav-dashboard');

            if (tabName === 'submit') {
                submitTab.classList.remove('hidden');
                dashboardTab.classList.add('hidden');
                navSubmit.classList.add('bg-white/20', 'shadow-sm');
                navDashboard.classList.remove('bg-white/20', 'shadow-sm');
            } else {
                submitTab.classList.add('hidden');
                dashboardTab.classList.remove('hidden');
                navDashboard.classList.add('bg-white/20', 'shadow-sm');
                navSubmit.classList.remove('bg-white/20', 'shadow-sm');
                loadWorks();
            }
        }

        // File label update helper
        function updateFileName(input) {
            const labelText = document.getElementById('file-label-text');
            if (input.files && input.files[0]) {
                labelText.innerHTML = `<span class="text-teal-600 font-semibold"><i class="fa-solid fa-file-arrow-up"></i> ${input.files[0].name}</span>`;
            } else {
                labelText.textContent = 'คลิกเพื่อเลือกไฟล์ หรือลากไฟล์มาวางที่นี่';
            }
        }

        // Custom Modal helper
        function showModal(title, message, isSuccess = true) {
            const modal = document.getElementById('custom-modal');
            const modalTitle = document.getElementById('modal-title');
            const modalMessage = document.getElementById('modal-message');
            const modalIcon = document.getElementById('modal-icon');

            modalTitle.textContent = title;
            modalMessage.textContent = message;

            if (isSuccess) {
                modalIcon.className = "w-16 h-16 rounded-full bg-emerald-50 text-emerald-600 flex items-center justify-center mx-auto mb-4 text-2xl shadow-inner";
                modalIcon.innerHTML = '<i class="fa-solid fa-check"></i>';
            } else {
                modalIcon.className = "w-16 h-16 rounded-full bg-rose-50 text-rose-600 flex items-center justify-center mx-auto mb-4 text-2xl shadow-inner";
                modalIcon.innerHTML = '<i class="fa-solid fa-triangle-exclamation"></i>';
            }

            modal.classList.remove('hidden');
            modal.classList.add('flex');
        }

        function closeModal() {
            const modal = document.getElementById('custom-modal');
            modal.classList.remove('flex');
            modal.classList.add('hidden');
        }

        // Save GAS URL to LocalStorage
        function saveGasUrl() {
            const url = document.getElementById('gasUrl').value.trim() || DEFAULT_GAS_URL;
            document.getElementById('gasUrl').value = url;
            localStorage.setItem('er_selaphum_gas_url', url);
            showModal('บันทึกสำเร็จ', 'บันทึก Google Apps Script Web App URL เรียบร้อยแล้ว');
        }

        // Load saved GAS URL on start
        window.addEventListener('DOMContentLoaded', () => {
            const savedUrl = localStorage.getItem('er_selaphum_gas_url') || DEFAULT_GAS_URL;
            document.getElementById('gasUrl').value = savedUrl;
            localStorage.setItem('er_selaphum_gas_url', savedUrl);
            loadWorks();
        });

            let fileDataName = '';
            let fileDataBase64 = '';
            let fileMimeType = '';

            if (fileInput.files && fileInput.files[0]) {
                const file = fileInput.files[0];
                fileDataName = file.name;
                fileMimeType = file.type;
                
                try {
                    fileDataBase64 = await readFileAsBase64(file);
                } catch (err) {
                    showModal('เกิดข้อผิดพลาด', 'ไม่สามารถอ่านไฟล์ได้ กรุณาลองใหม่อีกครั้ง', false);
                    submitBtn.disabled = false;
                    submitBtn.innerHTML = originalBtnText;
                    return;
                }
            }

            const newSubmission = {
                id: 'JOB-' + Math.floor(1000 + Math.random() * 9000),
                timestamp: timestamp,
                senderName: senderName,
                department: department,
                workTitle: workTitle,
                workDetail: workDetail,
                fileName: fileDataName,
                fileData: fileDataBase64,
                mimeType: fileMimeType,
                fileUrl: '#',
                status: 'รอตรวจสอบ',
                folderId: DRIVE_FOLDER_ID,
                driveFolderId: DRIVE_FOLDER_ID,
                driveFolderUrl: DRIVE_FOLDER_URL
            };

            const gasUrl = (localStorage.getItem('er_selaphum_gas_url') || DEFAULT_GAS_URL).trim();
            let submissionForStorage;

            try {
                const gasResult = await submitToGas(gasUrl, newSubmission);
                const uploadedFileUrl = extractFileUrl(gasResult);
                submissionForStorage = {
                    ...newSubmission,
                    fileUrl: uploadedFileUrl || newSubmission.fileUrl,
                    fileData: uploadedFileUrl ? '' : newSubmission.fileData
                };
            } catch (error) {
                console.error('GAS Error:', error);
                showModal('ส่งงานไม่สำเร็จ', error.message || 'ไม่สามารถส่งข้อมูลไปยัง Google Apps Script ได้', false);
                submitBtn.disabled = false;
                submitBtn.innerHTML = originalBtnText;
                return;
            }

            // Save to local storage for instant dashboard viewing
            let localWorks = JSON.parse(localStorage.getItem('er_selaphum_works') || '[]');
            localWorks.unshift(submissionForStorage);
            localStorage.setItem('er_selaphum_works', JSON.stringify(localWorks));

            submitBtn.disabled = false;
            submitBtn.innerHTML = originalBtnText;

            // Reset Form
            document.getElementById('submission-form').reset();
            document.getElementById('file-label-text').textContent = 'คลิกเพื่อเลือกไฟล์ หรือลากไฟล์มาวางที่นี่';

            showModal('ส่งงานสำเร็จ!', 'ข้อมูลและไฟล์ของคุณถูกบันทึกเข้าสู่ระบบเรียบร้อยแล้ว');
            switchTab('dashboard');
        }

        async function submitToGas(gasUrl, submission) {
            if (!gasUrl) {
                throw new Error('ยังไม่ได้ตั้งค่า Google Apps Script Web App URL');
            }

            const response = await fetch(gasUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'text/plain;charset=utf-8' },
                body: JSON.stringify(submission)
            });

            if (!response.ok) {
                throw new Error(`Google Apps Script ตอบกลับสถานะ ${response.status}`);
            }

            const responseText = await response.text();
            let result = {};
            if (responseText) {
                try {
                    result = JSON.parse(responseText);
                } catch (error) {
                    throw new Error(`Google Apps Script ส่งข้อมูลกลับมาไม่ถูกต้อง: ${responseText.slice(0, 120)}`);
                }
            }

            const status = String(result.status || '').toLowerCase();
            const rawFileUrl = result.fileUrl || result.fileURL || result.driveFileUrl || result.driveUrl || result.url || '';
            const message = result.message || result.error || rawFileUrl;
            if (status === 'error' || isGasErrorMessage(message)) {
                throw new Error(message || 'Google Apps Script ไม่สามารถบันทึกข้อมูลได้');
            }

            return result;
        }

        function extractFileUrl(result) {
            if (!result) return '';
            const fileUrl = result.fileUrl || result.fileURL || result.driveFileUrl || result.driveUrl || result.url || '';
            return isGasErrorMessage(fileUrl) || fileUrl === '-' ? '' : fileUrl;
        }

        function isGasErrorMessage(value) {
            const text = String(value || '');
            return /เกิดข้อผิดพลาด|ไม่ถูกต้อง|error|exception|invalid/i.test(text);
        }

        function readFileAsBase64(file) {
            return new Promise((resolve, reject) => {
                const reader = new FileReader();
                reader.onload = () => {
                    const result = reader.result;
                    const base64String = result.split(',')[1];
                    resolve(base64String);
                };
                reader.onerror = error => reject(error);
                reader.readAsDataURL(file);
            });
        }

        function getStoredWorks() {
            return JSON.parse(localStorage.getItem('er_selaphum_works') || '[]');
        }

        function saveStoredWorks(works) {
            localStorage.setItem('er_selaphum_works', JSON.stringify(works));
        }

        async function fetchWorksFromGas() {
            const gasUrl = (localStorage.getItem('er_selaphum_gas_url') || DEFAULT_GAS_URL).trim();
            if (!gasUrl) return [];

            const url = new URL(gasUrl);
            url.searchParams.set('action', 'list');
            url.searchParams.set('_', Date.now().toString());

            const response = await fetch(url.toString(), {
                method: 'GET',
                cache: 'no-store'
            });

            if (!response.ok) {
                throw new Error(`Google Apps Script ตอบกลับสถานะ ${response.status}`);
            }

            const result = await response.json();
            if (String(result.status || '').toLowerCase() === 'error') {
                throw new Error(result.error || 'ไม่สามารถโหลดข้อมูลเดิมจาก Google Apps Script ได้');
            }

            return Array.isArray(result.data) ? result.data : (Array.isArray(result.works) ? result.works : []);
        }

        function getWorkKey(work) {
            return work.id || `${work.timestamp || ''}|${work.senderName || ''}|${work.workTitle || ''}|${work.fileName || ''}`;
        }

        function mergeWorks(remoteWorks, localWorks) {
            const merged = new Map();

            [...remoteWorks, ...localWorks].forEach(work => {
                const key = getWorkKey(work);
                if (!key || merged.has(key)) return;
                merged.set(key, {
                    ...work,
                    fileUrl: work.fileUrl || '#',
                    fileData: work.fileData || '',
                    mimeType: work.mimeType || '',
                    status: work.status || 'รอตรวจสอบ'
                });
            });

            return Array.from(merged.values());
        }

        // Load and Render Dashboard Works & Charts
        async function loadWorks() {
            const localWorks = getStoredWorks();
            renderWorks(localWorks);

            try {
                const remoteWorks = await fetchWorksFromGas();
                if (remoteWorks.length > 0) {
                    const mergedWorks = mergeWorks(remoteWorks, localWorks);
                    saveStoredWorks(mergedWorks);
                    renderWorks(mergedWorks);
                }
            } catch (error) {
                console.warn('Load remote works failed:', error);
            }
        }

        function renderWorks(works) {
            const tbody = document.getElementById('work-table-body');
            const emptyState = document.getElementById('empty-state');

            // Update stats
            document.getElementById('stat-total').textContent = works.length;
            const approvedCount = works.filter(w => w.status === 'ตรวจสอบแล้ว').length;
            const pendingCount = works.filter(w => w.status === 'รอตรวจสอบ').length;
            document.getElementById('stat-approved').textContent = approvedCount;
            document.getElementById('stat-pending').textContent = pendingCount;

            const uniqueDepts = [...new Set(works.map(w => w.department))].filter(Boolean);
            document.getElementById('stat-dept-count').textContent = uniqueDepts.length;

            renderCharts(works);

            if (works.length === 0) {
                tbody.innerHTML = '';
                emptyState.classList.remove('hidden');
                return;
            }

            emptyState.classList.add('hidden');
            tbody.innerHTML = '';

            works.forEach((work, index) => {
                const tr = document.createElement('tr');
                tr.className = 'hover:bg-slate-50/80 transition';
                
                let statusBadge = '';
                if (work.status === 'ตรวจสอบแล้ว') {
                    statusBadge = `<span class="px-2.5 py-1 rounded-full text-[10px] font-semibold bg-emerald-50 text-emerald-600 border border-emerald-200">ตรวจสอบแล้ว</span>`;
                } else {
                    statusBadge = `<span class="px-2.5 py-1 rounded-full text-[10px] font-semibold bg-amber-50 text-amber-600 border border-amber-200">รอตรวจสอบ</span>`;
                }

                let fileAction = '-';
                if (work.fileName) {
                    if (work.fileData) {
                        fileAction = `<button onclick="downloadLocalFile(${index})" class="text-teal-600 hover:text-teal-800 font-medium underline flex items-center gap-1"><i class="fa-solid fa-download"></i> ดาวน์โหลด</button>`;
                    } else if (work.fileUrl && work.fileUrl !== '#') {
                        fileAction = `<a href="${work.fileUrl}" target="_blank" class="text-teal-600 hover:text-teal-800 font-medium underline flex items-center gap-1"><i class="fa-solid fa-arrow-up-right-from-square"></i> เปิดไฟล์</a>`;
                    } else {
                        fileAction = `<span class="text-slate-400">${work.fileName}</span>`;
                    }
                }

                tr.innerHTML = `
                    <td class="py-3.5 px-4 text-slate-500 whitespace-nowrap">${work.timestamp || '-'}</td>
                    <td class="py-3.5 px-4">
                        <div class="font-medium text-slate-900">${work.senderName}</div>
                        <div class="text-[11px] text-slate-400">${work.department}</div>
                    </td>
                    <td class="py-3.5 px-4">
                        <div class="font-medium text-slate-900">${work.workTitle}</div>
                        <div class="text-[11px] text-slate-400 truncate max-w-xs">${work.workDetail || ''}</div>
                    </td>
                    <td class="py-3.5 px-4 whitespace-nowrap">${statusBadge}</td>
                    <td class="py-3.5 px-4 whitespace-nowrap">${fileAction}</td>
                    <td class="py-3.5 px-4 text-center whitespace-nowrap">
                        <button onclick="toggleStatus(${index})" class="px-2.5 py-1 bg-slate-100 hover:bg-slate-200 text-slate-700 rounded-lg text-[11px] transition font-medium" title="เปลี่ยนสถานะ">เปลี่ยนสถานะ</button>
                        <button onclick="deleteWork(${index})" class="px-2.5 py-1 bg-rose-50 hover:bg-rose-100 text-rose-600 rounded-lg text-[11px] transition ml-1" title="ลบข้อมูล"><i class="fa-solid fa-trash"></i></button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        // Render Chart.js Analytics
        function renderCharts(works) {
            const positions = ["พยาบาล", "เจ้าพนักงานเวชกิจฉุกเฉิน", "ผู้ช่วยเหลือคนไข้", "บันทึกข้อมูล"];
            const posCounts = positions.map(p => works.filter(w => w.department === p).length);

            // Bar Chart
            const deptCtx = document.getElementById('deptChart').getContext('2d');
            if (deptChartInstance) deptChartInstance.destroy();
            deptChartInstance = new Chart(deptCtx, {
                type: 'bar',
                data: {
                    labels: positions,
                    datasets: [{
                        label: 'จำนวนงานที่ส่ง',
                        data: posCounts,
                        backgroundColor: '#0d9488',
                        borderRadius: 10
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: { legend: { display: false } },
                    scales: {
                        y: { beginAtZero: true, ticks: { precision: 0 } }
                    }
                }
            });

            // Doughnut Chart
            const approved = works.filter(w => w.status === 'ตรวจสอบแล้ว').length;
            const pending = works.filter(w => w.status === 'รอตรวจสอบ').length;

            const statusCtx = document.getElementById('statusChart').getContext('2d');
            if (statusChartInstance) statusChartInstance.destroy();
            statusChartInstance = new Chart(statusCtx, {
                type: 'doughnut',
                data: {
                    labels: ['ตรวจสอบแล้ว', 'รอตรวจสอบ'],
                    datasets: [{
                        data: [approved, pending],
                        backgroundColor: ['#10b981', '#f59e0b'],
                        borderWidth: 0
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: { position: 'bottom', labels: { boxWidth: 12, font: { family: 'Prompt' } } }
                    }
                }
            });
        }

        // Download local base64 file
        function downloadLocalFile(index) {
            const works = JSON.parse(localStorage.getItem('er_selaphum_works') || '[]');
            const work = works[index];
            if (!work || !work.fileData) return;

            const link = document.createElement('a');
            link.href = `data:${work.mimeType};base64,${work.fileData}`;
            link.download = work.fileName;
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }

        // Toggle Status
        function toggleStatus(index) {
            let works = getStoredWorks();
            if (works[index]) {
                works[index].status = works[index].status === 'รอตรวจสอบ' ? 'ตรวจสอบแล้ว' : 'รอตรวจสอบ';
                saveStoredWorks(works);
                renderWorks(works);
            }
        }

        // Delete Work
        function deleteWork(index) {
            if (confirm('คุณต้องการลบรายการนี้ใช่หรือไม่?')) {
                let works = getStoredWorks();
                works.splice(index, 1);
                saveStoredWorks(works);
                renderWorks(works);
            }
        }

        // Search Filter Table
        function filterTable() {
            const input = document.getElementById('search-input').value.toLowerCase();
            const rows = document.getElementById('work-table-body').getElementsByTagName('tr');

            for (let i = 0; i < rows.length; i++) {
                const text = rows[i].textContent.toLowerCase();
                if (text.includes(input)) {
                    rows[i].style.display = "";
                } else {
                    rows[i].style.display = "none";
                }
            }
        }
